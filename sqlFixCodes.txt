
START TRANSACTION;
UPDATE users u
JOIN (
    SELECT 
        id, 
        UPPER(SUBSTRING(last_name, 1, 3)) AS prefix,
        ROW_NUMBER() OVER (
            PARTITION BY UPPER(SUBSTRING(last_name, 1, 3)) 
            ORDER BY id
        ) AS sequence_num
    FROM users
) su ON u.id = su.id
SET u.code = CONCAT(su.prefix, LPAD(su.sequence_num, 4, '0'));
 
-- This command permanently writes the changes to your database
COMMIT;
